    <script>

        (function() {

            // 目标时间: 2026-10-12 19:00:00 北京时间 (UTC+8)

            const targetDate = new Date('2026-10-12T19:00:00+08:00').getTime();

 

            // 倒计时元素

            const daysEl = document.getElementById('days');

            const hoursEl = document.getElementById('hours');

            const minutesEl = document.getElementById('minutes');

            const secondsEl = document.getElementById('seconds');

 

            // 乱码标题元素

            const exTitleEl = document.getElementById('exTitle');

 

            // 原始目标文本

            const TARGET_TEXT = 'Ex Light & Bass 2026';

            // 乱码字符集 —— 使用类似终端的字符，视觉上符合风格

            const GARBAGE_CHARS = '!@#$%^&*()_+{}:"<>?|/\\~`±§¥£ΩΔΘΛΞΠΣΦΨΩαβγδεζηθικλμνξοπρστυφχψω0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ';

           

            // 计算总进度: 从 2026-10-08 00:00:00 到 2026-10-12 19:00:00

            const startDate = new Date('2026-10-08T00:00:00+08:00').getTime();

            const endDate = targetDate;

            const totalDuration = endDate - startDate;

 

            // 存储最新的乱码字符串

            let currentGarbageText = '';

 

            // 生成随机乱码 (长度与目标文本相同)

            function generateGarbage(length) {

                let result = '';

                for (let i = 0; i < length; i++) {

                    // 稍微增加一些空格概率，让乱码看起来更自然

                    if (Math.random() < 0.15) {

                        result += ' ';

                    } else {

                        const randomIndex = Math.floor(Math.random() * GARBAGE_CHARS.length);

                        result += GARBAGE_CHARS[randomIndex];

                    }

                }

                return result;

            }

 

            // 计算当前进度 (0~1)，基于当前时间与总时间区间的比较

            function getProgress() {

                const now = Date.now();

                if (now >= endDate) return 1;

                if (now <= startDate) return 0;

                return (now - startDate) / totalDuration;

            }

 

            // 根据进度生成显示的标题字符串：乱码逐渐变为目标文本

            function getTitleByProgress(progress) {

                const targetLength = TARGET_TEXT.length;

                let result = '';

               

                // 对于每个字符位置，根据进度决定是显示目标字符还是随机乱码

                // 使用全局进度加上一点随机偏移会让每个字符的显现时刻略有不同，看起来更自然

                // 但为了“按照倒计时进度逐渐显示”，我们用 progress 作为基础阈值

                // 但为了让过渡更平滑，每个字符的显现阈值可以在 progress 附近随机波动 (但必须保证最终全部显示)

                for (let i = 0; i < targetLength; i++) {

                    // 每个字符有一个阈值: 基础阈值随索引增加而略微增加，也可以基于字符位置

                    // 但我们希望进度到达1时全部显示，进度0时全是乱码。

                    // 这里采用: 字符显现的阈值 = i / targetLength * 0.8 + 0.1? 但那样可能最后一部分字符在进度0.9之后才显现。

                    // 但我们的progress会增长到1，所以没问题。

                    // 更直观：每一个字符的显现进度阈值 = (i+1)/targetLength * 0.9? 会留下最后10%确保全部显现

                    // 但我们希望进度1时完全显现，所以阈值最大 <=1。

                    // 设定: threshold = (i / targetLength) * 0.9 + 0.05  (范围0.05~0.95)

                    // 然后加一点小随机使其错落有致: 但随机可能造成某些字符提前显现，没问题。

                    // 简单方法: 使用 index / targetLength 作为基本阈值，再乘以0.95，并加上0.02*随机

                    const baseThreshold = (i / targetLength) * 0.9 + 0.05; // 0.05 ~ 0.95

                    // 为了让抖动自然，每个字符额外加上一个固定的微小偏移 (基于字符索引的伪随机)

                    const randomOffset = (Math.sin(i * 100) * 0.05); // -0.05 ~ 0.05

                    let threshold = baseThreshold + randomOffset;

                    // 钳制在0~1之间

                    threshold = Math.min(1, Math.max(0, threshold));

                   

                    // 如果进度大于阈值，显示目标字符

                    if (progress > threshold) {

                        result += TARGET_TEXT[i];

                    } else {

                        // 否则显示乱码字符 (随机)

                        result += GARBAGE_CHARS[Math.floor(Math.random() * GARBAGE_CHARS.length)];

                    }

                }

                // 但为了让乱码“持续变化”，即使全部显现后也可以有微小抖动？不，全部显现后就固定了。

                // 但根据需求“按照倒计时进度将乱码逐渐显示成Ex Light & Bass 2026”，全部显现后就不需要乱码了。

                // 但如果我们希望标题看起来更动态，可以在progress=1后停止更新？但setInterval还是会继续跑，但文本已固定。

                // 我们让定时器每秒/每帧更新，但在progress=1后就不再改变目标文本。其实没关系。

                return result;

            }

 

            // 倒计时更新

            function updateCountdown() {

                const now = Date.now();

                const diff = targetDate - now;

 

                if (diff <= 0) {

                    // 倒计时结束

                    daysEl.textContent = '00';

                    hoursEl.textContent = '00';

                    minutesEl.textContent = '00';

                    secondsEl.textContent = '00';

                    // 确保乱码完全显现（进度为1）

                    exTitleEl.textContent = TARGET_TEXT;

                    return;

                }

 

                const days = Math.floor(diff / (1000 * 60 * 60 * 24));

                const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

                const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));

                const seconds = Math.floor((diff % (1000 * 60)) / 1000);

 

                daysEl.textContent = days.toString().padStart(2, '0');

                hoursEl.textContent = hours.toString().padStart(2, '0');

                minutesEl.textContent = minutes.toString().padStart(2, '0');

                secondsEl.textContent = seconds.toString().padStart(2, '0');

            }

 

            // 更新乱码标题 (基于进度)

            function updateExTitle() {

                const progress = getProgress(); // 0~1

                // 如果进度 >= 1，直接显示目标文本（已经全部显现）

                if (progress >= 1) {

                    exTitleEl.textContent = TARGET_TEXT;

                    return;

                }

                // 否则生成混合文本

                const mixed = getTitleByProgress(progress);

                exTitleEl.textContent = mixed;

            }

 

            // 乱码持续变化：即使progress为0，也需要不断变化乱码。使用单独的定时器高频更新乱码部分。

            // 但注意: 如果我们在updateExTitle中每次重新生成乱码，由于progress变化缓慢，混合文本会逐渐显现，但乱码部分也会变化。

            // 但为了“持续变化的乱码”，我们需要乱码部分不断刷新，而已经显现的文字保持不变。

            // 这就需要更精细的控制：存储每个字符是否已经变为目标字符的状态，但用progress来决定什么时候切换可能更好。

            // 但progress变化太慢，乱码部分变化频率可以更快。我们可以每帧用当前进度重新生成混合文本，

            // 但这样乱码字符会随机变化，而目标字符一旦出现就不会再变。这符合要求。

            // 但是每帧都调用getTitleByProgress会导致已经显现的字符保持不变，而乱码部分不断变化。完美！

            // 但需要注意：如果progress接近1，可能大部分字符已经显现，但少数乱码依然变化。

            // 使用requestAnimationFrame可以达到60fps变化。

           

            let lastTimestamp = 0;

            const FPS_INTERVAL = 1000 / 30; // 限制为30fps，避免性能问题，也足够乱码闪烁

           

            function animateTitle(timestamp) {

                if (!lastTimestamp) lastTimestamp = timestamp;

                const elapsed = timestamp - lastTimestamp;

               

                if (elapsed > FPS_INTERVAL) {

                    // 更新标题（乱码变化）

                    updateExTitle();

                    lastTimestamp = timestamp;

                }

               

                // 继续下一帧

                requestAnimationFrame(animateTitle);

            }

 

            // 启动倒计时定时器 (每秒更新一次)

            updateCountdown();

            setInterval(updateCountdown, 1000);

 

            // 启动乱码动画 (使用requestAnimationFrame)

            requestAnimationFrame(animateTitle);

 

            // 初始化一次标题，防止空白

            updateExTitle();

 

            // 注意：如果用户系统时间不在2026-10-08到2026-10-12之间，倒计时可能显示为0或者进度不对。

            // 但我们的目标区间是固定的，进度从2026-10-08开始计算，所以如果当前时间早于2026-10-08，进度为0，标题全是乱码，倒计时还有很久。

            // 如果当前时间晚于2026-10-12 19:00，进度为1，标题显示目标文字，倒计时归零。

            // 符合需求。

        })();

    </script>
